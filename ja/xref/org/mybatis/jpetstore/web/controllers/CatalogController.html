16  package org.mybatis.jpetstore.web.controllers;
17  
18  import java.util.List;
19  
20  import org.mybatis.jpetstore.domain.Category;
21  import org.mybatis.jpetstore.domain.Item;
22  import org.mybatis.jpetstore.domain.Product;
23  import org.mybatis.jpetstore.service.CatalogService;
24  import org.springframework.beans.factory.annotation.Autowired;
25  import org.springframework.stereotype.Controller;
26  import org.springframework.ui.Model;
27  import org.springframework.web.bind.annotation.GetMapping;
28  import org.springframework.web.bind.annotation.RequestMapping;
29  import org.springframework.web.bind.annotation.RequestParam;
30  
31  @Controller
32  @RequestMapping("/catalog")
33  public class CatalogController {
34  
35    private static final String ERROR_VIEW = "common/Error";
36  
37    @Autowired
38    private CatalogService catalogService;
39  
40    @GetMapping({ "", "/" })
41    public String viewMain() {
42      return "catalog/Main";
43    }
44  
45    @GetMapping("/viewCategory")
46    public String viewCategory(@RequestParam(value = "categoryId", required = false) String categoryId, Model model) {
47      if (categoryId != null) {
48        List<Product> productList = catalogService.getProductListByCategory(categoryId);
49        Category category = catalogService.getCategory(categoryId);
50        model.addAttribute("productList", productList);
51        model.addAttribute("category", category);
52      }
53      return "catalog/Category";
54    }
55  
56    @GetMapping("/viewProduct")
57    public String viewProduct(@RequestParam(value = "productId", required = false) String productId, Model model) {
58      if (productId != null) {
59        List<Item> itemList = catalogService.getItemListByProduct(productId);
60        Product product = catalogService.getProduct(productId);
61        model.addAttribute("itemList", itemList);
62        model.addAttribute("product", product);
63      }
64      return "catalog/Product";
65    }
66  
67    @GetMapping("/viewItem")
68    public String viewItem(@RequestParam("itemId") String itemId, Model model) {
69      Item item = catalogService.getItem(itemId);
70      Product product = item.getProduct();
71      model.addAttribute("item", item);
72      model.addAttribute("product", product);
73      return "catalog/Item";
74    }
75  
76    @GetMapping("/searchProducts")
77    public String searchProducts(@RequestParam(value = "keyword", required = false) String keyword, Model model) {
78      if (keyword == null || keyword.isEmpty()) {
79        model.addAttribute("message", "Please enter a keyword to search for, then press the search button.");
80        return ERROR_VIEW;
81      }
82      List<Product> productList = catalogService.searchProductList(keyword.toLowerCase());
83      model.addAttribute("productList", productList);
84      return "catalog/SearchProducts";
85    }
86  
87  }
