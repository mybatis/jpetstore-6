53    @Test
54    void viewCartReturnsCartView() {
55      HttpSession session = mock(HttpSession.class);
56      Cart cart = new Cart();

80    /**
81     * Add item with new id fetches item from catalog and adds to cart.
82     */
83    @Test
84    void addItemWithNewIdFetchesItemFromCatalogAndAddsToCart() {
85      HttpSession session = mock(HttpSession.class);
86      Model model = new ExtendedModelMap();
87  
88      Cart cart = new Cart();
89      when(session.getAttribute("cart")).thenReturn(cart);
90  
91      Item item = new Item();
92      item.setItemId("EST-1");
93      item.setListPrice(new java.math.BigDecimal("10.00"));
94      when(catalogService.isItemInStock("EST-1")).thenReturn(true);
95      when(catalogService.getItem("EST-1")).thenReturn(item);
96  
97      String view = cartController.addItemToCart("EST-1", session, model);
98  
99      assertThat(view).isEqualTo("cart/Cart");
100     assertThat(cart.containsItemId("EST-1")).isTrue();
101     assertThat(cart.getCartItemList().get(0).getQuantity()).isEqualTo(1);
102     assertThat(model.asMap().get("cart")).isSameAs(cart);
103   }
