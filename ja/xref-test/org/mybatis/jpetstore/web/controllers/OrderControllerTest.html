137   /**
138    * Confirm order uses billing address from session.
139    */
140   @Test
141   void confirmOrderUsesBillingAddressFromSession() {
142     // When the ConfirmOrder form is submitted it only posts confirmed=true.
143     // The @ModelAttribute Order is therefore all-null. The controller must use
144     // the session order (which has the full billing address) and must NOT
145     // overwrite it with the null-filled model attribute.
146     HttpSession session = mock(HttpSession.class);
147     Model model = new ExtendedModelMap();
148 
149     org.mybatis.jpetstore.domain.Order sessionOrder = new org.mybatis.jpetstore.domain.Order();
150     sessionOrder.setBillAddress1("901 San Antonio Road");
151     sessionOrder.setBillCity("Palo Alto");
152     sessionOrder.setBillState("CA");
153     sessionOrder.setBillZip("94303");
154     sessionOrder.setBillCountry("USA");
155     sessionOrder.setBillToFirstName("ABC");
156     sessionOrder.setBillToLastName("Banner");
157     sessionOrder.setCreditCard("999 9999 9999 9999");
158     sessionOrder.setExpiryDate("12/03");
159     sessionOrder.setCardType("Visa");
160 
161     when(session.getAttribute("order")).thenReturn(sessionOrder);
162 
163     // Simulate ConfirmOrder POST: empty @ModelAttribute (all fields null), confirmed=true
164     org.mybatis.jpetstore.domain.Order emptyOrder = new org.mybatis.jpetstore.domain.Order();
165     String view = orderController.newOrder(emptyOrder, false, true, session, model);
166 
167     assertThat(view).isEqualTo("order/ViewOrder");
168     assertThat(model.asMap()).containsEntry("message", "Thank you, your order has been submitted.");
169 
170     // Verify insertOrder was called exactly once
171     verify(orderService, times(1)).insertOrder(any(org.mybatis.jpetstore.domain.Order.class));
172 
173     // The session order must NOT have been overwritten with null
174     assertThat(sessionOrder.getBillAddress1()).isEqualTo("901 San Antonio Road");
175     assertThat(sessionOrder.getBillCity()).isEqualTo("Palo Alto");
176   }

178   /**
179    * Shipping form submission does not overwrite billing fields.
180    */
181   @Test
182   void shippingFormSubmissionDoesNotOverwriteBillingFields() {
183     // ShippingForm only posts shipping fields (shipToFirstName etc.) + confirmed=false.
184     // The billing/payment fields in @ModelAttribute Order are null (not in the form).
185     // The controller must NOT overwrite the billing address stored in the session
186     // with these null values, or the subsequent insertOrder will fail with a
187     // NOT NULL constraint violation on BILLADDR1.
188     HttpSession session = mock(HttpSession.class);
189     Model model = new ExtendedModelMap();
190 
191     org.mybatis.jpetstore.domain.Order sessionOrder = new org.mybatis.jpetstore.domain.Order();
192     sessionOrder.setBillAddress1("901 San Antonio Road");
193     sessionOrder.setBillCity("Palo Alto");
194     sessionOrder.setBillState("CA");
195     sessionOrder.setBillZip("94303");
196     sessionOrder.setBillCountry("USA");
197     sessionOrder.setBillToFirstName("ABC");
198     sessionOrder.setBillToLastName("Banner");
199     sessionOrder.setCreditCard("999 9999 9999 9999");
200     sessionOrder.setExpiryDate("12/03");
201     sessionOrder.setCardType("Visa");
202 
203     when(session.getAttribute("order")).thenReturn(sessionOrder);
204 
205     // Simulate ShippingForm POST: only shipping fields present, billing all null
206     org.mybatis.jpetstore.domain.Order shippingFormOrder = new org.mybatis.jpetstore.domain.Order();
207     shippingFormOrder.setShipToFirstName("ABC");
208     shippingFormOrder.setShipToLastName("Banner");
209     shippingFormOrder.setShipAddress1("901 San Antonio Road");
210     shippingFormOrder.setShipAddress2("MS UCUP02-207");
211     shippingFormOrder.setShipCity("Palo Alto");
212     shippingFormOrder.setShipState("CA");
213     shippingFormOrder.setShipZip("94303");
214     shippingFormOrder.setShipCountry("USA");
215     // billing/payment fields left null (not submitted by ShippingForm)
216 
217     String view = orderController.newOrder(shippingFormOrder, false, false, session, model);
218 
219     assertThat(view).isEqualTo("order/ConfirmOrder");
220 
221     // Billing fields in the session order must still have their original values
222     assertThat(sessionOrder.getBillAddress1()).isEqualTo("901 San Antonio Road");
223     assertThat(sessionOrder.getBillCity()).isEqualTo("Palo Alto");
224     assertThat(sessionOrder.getCreditCard()).isEqualTo("999 9999 9999 9999");
225     assertThat(sessionOrder.getCardType()).isEqualTo("Visa");
226 
227     // Shipping fields must have been updated from the form
228     assertThat(sessionOrder.getShipAddress2()).isEqualTo("MS UCUP02-207");
229   }
